//@ pack Blocks Editor (UI)
//@ target ui
// Turns the Game tab's code editor into a visual block editor, a bit like Scratch.
//
//   Palette      The tray at the bottom has the block kinds, in groups. Drag one onto the canvas,
//                or just tap it: it joins the block you picked, or lands in the middle of the canvas.
//   Canvas       An endless canvas that grows in every direction. Drag empty space to move around, pinch
//                or Ctrl + scroll to zoom. Each game tab (Main and the others) has its own canvas.
//   Blocks       Drag a block to move it, with the blocks under it. Blocks that wrap code (like if / else)
//                grow with what is inside them. Drop a round block into a round slot of another block.
//                Drop a block on the palette to throw it away, or tap it and use Delete.
//   Buttons      Clean up lines the stacks up, Fit all shows everything, Center goes back to the start.
//   Hash         Shows the code the blocks make. The game runs that code.
//   Errors       When the game has an error, the block that made it lights up red.
//
// Blocks replace the text editor for the Game tab. The Engine, Compiler, UI, Save and Packer code is all
// still there, only hidden. File > Blocks > Show text code brings the normal editor back. Text that is
// typed in the editor is replaced the next time you change the blocks.
// The blocks are saved on the first line of each game tab (a note that begins //@ blocks), so they go
// with the project and with Export game. Code that was already in a tab starts out as one Raw Hash block.
// After you turn the pack on, it starts working the next time Abyss Engine loads.
//
// Dependencies: Blocks Core (UI). Turn it On first. Blocks Basic (UI) gives the blocks to use.
// This pack only adds things. Every name it adds starts with bk.

#bkMk(tag, cls)[
  {el} = make(tag)
  el.className = cls

// Sizes a text field to fit what is in it.
#bkFit(el, multi)[
  {v} = el.value
  @(multi)[
    {ls} = split(v, "\n")
    {w} = 6
    %(ls, l)[ w = max(w, add(size(l), 1)) ]
    el.cols = w
    el.rows = size(ls)
  ][
    {n} = size(v)
    @(eq(n, 0))[ n = size(el.placeholder) ]
    el.size = max(2, add(n, 1))
  ]
]

// One input of a block: the block sitting in it, or a text field.
#bkInputEl(b, d, nm, pal)[
  @(not(has(b.args, nm)))[ put(b.args, nm, bkTextArg()) ]
  {a} = get(b.args, nm)
  {box} = bkMk("span", "bk-in")
  box.dataset.owner = b.id
  box.dataset.arg = nm
  @(has(a, "b"))[ box.appendChild(bkBlockEl(a.b, pal)) ][
    {multi} = has(d.multiline, nm)
    {tag} = "input"
    @(multi)[ tag = "textarea" ]
    {inp} = bkMk(tag, "bk-field")
    @(not(multi))[ inp.type = "text" ]
    inp.spellcheck = false
    inp.setAttribute("autocapitalize", "off")
    inp.setAttribute("autocomplete", "off")
    @(has(d.defaults, nm))[ inp.placeholder = get(d.defaults, nm) ][ inp.placeholder = "0" ]
    inp.value = a.t
    bkFit(inp, multi)
    @(and(eq(nm, "name"), d.dynCall))[ inp.setAttribute("list", "bkFns") ]
    @(and(eq(nm, "cat"), d.hasParams))[ inp.setAttribute("list", "bkCatList") ]
    @(pal)[ inp.disabled = true ][
      on(inp, "input", #(e)[
        a.t = inp.value
        bkFit(inp, multi)
        bkTouched()
      ])
      // Calls look up their function by this name, so the blocks are drawn again once it is set.
      @(and(or(eq(nm, "name"), eq(nm, "cat")), or(d.dynCall, d.hasParams)))[ on(inp, "change", #(e)[ bkChanged() ]) ]
    ]
    box.appendChild(inp)
  ]
  box
]

// The words and inputs along a block's top row.
#bkParts(b, d, parent, pal)[
  %(d.parts, p)[
    @(eq(slice(p, 0, 1), "$"))[ parent.appendChild(bkInputEl(b, d, slice(p, 1, size(p)), pal)) ][
      {t} = bkMk("span", "bk-t")
      t.textContent = p
      parent.appendChild(t)
    ]
  ]
  @(d.hasParams)[ bkParamEls(b, parent, pal) ]
  @(d.dynCall)[ bkCallEls(b, d, parent, pal) ]
]

// Named inputs on a definition block, with x to remove one and + to add one.
#bkParamEls(b, parent, pal)[
  {ps} = bkParamsOf(b)
  %(size(ps), i)[
    {k} = i
    {box} = bkMk("span", "bk-in")
    {inp} = bkMk("input", "bk-field bk-param")
    inp.type = "text"
    inp.spellcheck = false
    inp.setAttribute("autocapitalize", "off")
    inp.placeholder = "name"
    inp.value = get(ps, k)
    bkFit(inp, false)
    @(pal)[ inp.disabled = true ][
      on(inp, "input", #(e)[
        put(b.params, k, inp.value)
        bkFit(inp, false)
        bkTouched()
      ])
      on(inp, "change", #(e)[ bkChanged() ])
    ]
    box.appendChild(inp)
    @(not(pal))[
      {x} = bkMk("button", "bk-mini")
      x.type = "button"
      x.textContent = "×"
      x.setAttribute("aria-label", "Remove input")
      on(x, "mousedown", #(e)[ e.preventDefault() ])
      on(x, "pointerdown", #(e)[
        e.preventDefault()
        removeAt(b.params, k)
        bkChanged()
      ])
      box.appendChild(x)
    ]
    parent.appendChild(box)
  ]
  {plus} = bkMk("button", "bk-mini")
  plus.type = "button"
  plus.textContent = "+"
  plus.setAttribute("aria-label", "Add an input")
  @(not(pal))[
    on(plus, "mousedown", #(e)[ e.preventDefault() ])
    on(plus, "pointerdown", #(e)[
      e.preventDefault()
      bkAddParam(b)
    ])
  ]
  parent.appendChild(plus)
  // Says whether the function gives an answer (then it is a round block) or just does steps.
  {rt} = bkMk("button", "bk-mini bk-ret")
  rt.type = "button"
  rt.textContent = "answer"
  @(bkDefIsAnswer(b))[ rt.textContent = "answer ✓" ]
  @(not(pal))[
    on(rt, "mousedown", #(e)[ e.preventDefault() ])
    on(rt, "pointerdown", #(e)[
      e.preventDefault()
      b.ret = not(bkDefIsAnswer(b))
      bkChanged()
    ])
  ]
  parent.appendChild(rt)
]

// Adds a new named input to a definition and puts the cursor in its name.
#bkAddParam(b)[
  @(not(has(b, "params")))[ b.params = list() ]
  {n} = add(size(b.params), 1)
  {nm} = join("input", n)
  %(has(b.params, nm))[
    n = add(n, 1)
    nm = join("input", n)
  ]
  push(b.params, nm)
  bkChanged()
  later(30, #[
    {els} = domAll(join("#bkWorld [data-id=", b.id, "] > .bk-head .bk-param"))
    @(size(els))[
      {el} = get(els, sub(size(els), 1))
      el.focus()
      el.select()
    ]
  ])
]

// A call block is the function's name, then one input for each input the function has. With no such function the
// name is a box (so it can be fixed) and there is one free text.
#bkCallEls(b, d, parent, pal)[
  {def} = bkFnDefFor(bkWs, bkArgText(b, d, "name"))
  @(def)[
    {n} = bkMk("span", "bk-t")
    n.textContent = bkArgText(b, d, "name")
    parent.appendChild(n)
    {ps} = bkParamsOf(def)
    %(size(ps), i)[
      {t} = bkMk("span", "bk-t")
      t.textContent = get(ps, i)
      parent.appendChild(t)
      parent.appendChild(bkInputEl(b, d, join("p", i), pal))
    ]
  ][
    {c} = bkMk("span", "bk-t")
    c.textContent = "call"
    parent.appendChild(c)
    parent.appendChild(bkInputEl(b, d, "name", pal))
    {t} = bkMk("span", "bk-t")
    t.textContent = "with"
    parent.appendChild(t)
    parent.appendChild(bkInputEl(b, d, "args", pal))
  ]
]

// The color of a block: functions take the color of the category they were put in.
#bkBlockColor(b)[
  {d} = bkKind(b.type)
  {c} = bkColor(b.type)
  @(d)[
    @(d.hasParams)[ c = bkCatColorFor(bkDefCat(b)) ]
    @(d.dynCall)[
      {def} = bkFnDefFor(bkWs, bkArgText(b, d, "name"))
      @(def)[ c = bkCatColorFor(bkDefCat(def)) ]
    ]
  ]
  c
]

      el = bkMk("div", "bk-block bk-wrap")

#bkCatHas(cat)[

#bkRenderPal[
  bkChips.innerHTML = ""
  bkPalItems.innerHTML = ""
  @(not(bkCatHas(bkCat)))[
    bkCat = ""
    %(bkCats, c)[ @(and(eq(bkCat, ""), bkCatHas(c)))[ bkCat = c ] ]
  ]
  %(bkCats, c)[
    @(bkCatHas(c))[
      {name} = c
      {b} = make("button")
      b.type = "button"
      b.textContent = name
      b.style.setProperty("--c", get(bkCatColor, name))
      b.dataset.on = "no"
      @(eq(name, bkCat))[ b.dataset.on = "yes" ]
      on(b, "click", #(e)[
        bkCat = name
        bkRenderPal()
      ])
      bkChips.appendChild(b)
    ]
  ]
  %(bkOrder, t)[
    @(eq(get(bkReg, t).cat, bkCat))[
      @(not(get(bkReg, t).hidden))[ bkPalItems.appendChild(bkBlockEl(bkMake(bkPalWs, t), true)) ]
    ]
  ]
  // Every function put in this category is a block here, like any other.
  %(bkFnDefs, df)[
    @(eq(bkDefCat(df), bkCat))[
      {nm} = bkDefName(df)
      {kind} = "fn_call"
      @(bkDefIsAnswer(df))[ kind = "fn_value" ]
      {cb} = bkMake(bkPalWs, kind)
      {na} = map()
      na.t = nm
      put(cb.args, "name", na)
      {el} = bkBlockEl(cb, true)
      el.dataset.fn = nm
      bkPalItems.appendChild(el)
    ]
  ]
]

// Draws the palette again when the functions have changed (new, renamed, moved to another category).
#bkSyncPal[
  bkFnDefs = bkAllDefs(bkWs)
  {sig} = list()
  %(bkFnDefs, df)[ push(sig, join(bkDefName(df), "|", bkDefCat(df), "|", joinList(bkParamsOf(df), ","), "|", bkDefIsAnswer(df))) ]
  {s} = joinList(sig, ";")
  @(neq(s, bkPalSig))[
    bkPalSig = s

// Puts items into a list at a place, keeping the same list.
#bkSplice(l, at, items)[
  {tail} = list()
  %(sub(size(l), at))[ push(tail, removeAt(l, at)) ]
  %(items, it)[ push(l, it) ]
  %(tail, it)[ push(l, it) ]
  true
]

// Where is this block? Gives {list, idx} for a block in a list of steps, {owner, arg} for a block in an input, or false.
#bkLocIn(blocks, id)[
  {hit} = false
  %(size(blocks), i)[
    @(not(hit))[
      {b} = get(blocks, i)
      @(eq(b.id, id))[
        hit = map()
        hit.list = blocks
        hit.idx = i
      ][
        %(keys(b.args), nm)[
          {a} = get(b.args, nm)
          @(not(hit))[
            @(has(a, "b"))[
              @(eq(a.b.id, id))[
                hit = map()
                hit.owner = b
                hit.arg = nm
              ][ hit = bkLocIn(list(a.b), id) ]
            ]
          ]
        ]
        %(b.slots, body)[ @(not(hit))[ hit = bkLocIn(body, id) ] ]
      ]
    ]
  ]
  hit
]

#bkLocate(id)[
  {hit} = false
  %(bkWs.stacks, st)[ @(not(hit))[ hit = bkLocIn(st.blocks, id) ] ]
  hit
]

#bkTextArg[
  {a} = map()
  a.t = ""
  a
]

// Stacks with nothing in them go away.
#bkTidy[
  {keepers} = list()
  %(bkWs.stacks, st)[ @(gt(size(st.blocks), 0))[ push(keepers, st) ] ]
  bkWs.stacks = keepers
  true
]

// Takes a block out of the blocks, along with the steps under it. Gives back the list of what came out.
#bkCut(id)[
  {loc} = bkLocate(id)
  {moved} = list()
  @(loc)[
    @(has(loc, "list"))[
      %(sub(size(loc.list), loc.idx))[ push(moved, removeAt(loc.list, loc.idx)) ]
    ][
      push(moved, get(get(loc.owner.args, loc.arg), "b"))
      put(loc.owner.args, loc.arg, bkTextArg())
    ]
  ]
  bkTidy()
  moved
]

#bkAddStack(blocks, x, y)[
  {st} = map()
  st.x = round(x)
  st.y = round(y)
  st.blocks = blocks
  push(bkWs.stacks, st)
  st
]

  {sel} = bkSelected()
  {done} = false
  @(bkIsValue(type))[
    @(sel)[
      {sd} = bkKind(sel.type)
      @(sd)[
        %(sd.inputs, nm)[
          @(not(done))[
            @(not(has(get(sel.args, nm), "b")))[
              {slotArg} = map()
              slotArg.b = b
              put(sel.args, nm, slotArg)
              done = true
            ]
          ]
        ]
      ]
    ]
  ][
    @(sel)[
      @(bkIsStep(sel.type))[
        {sd} = bkKind(sel.type)
        @(and(sd, eq(get(sd, "shape"), "wrap")))[

  ]
  @(not(done))[
    {r} = bkCanvas.getBoundingClientRect()

    bkAddStack(list(b), add(sub(bkWorldX(add(r.left, div(r.width, 2))), 60), off), add(sub(bkWorldY(add(r.top, div(r.height, 2))), 30), off))

// What is under the pointer while blocks are carried: where they would land.
#bkTarget(x, y, first)[
  {t} = map()
  t.kind = "none"
  {el} = doc().elementFromPoint(x, y)
  @(el)[
    @(el.closest(".bk-pal"))[ t.kind = "trash" ][
      @(el.closest("#bkCanvas"))[
        t.kind = "canvas"
        {shape} = bkShape(first.type)
        @(bkIsValue(first.type))[
          {f} = el.closest(".bk-in")
          @(f)[
            t.kind = "input"
            t.el = f
            t.owner = f.dataset.owner
            t.arg = f.dataset.arg
          ]
        ][
          @(true)[
            {sl} = el.closest(".bk-slot")
            {bl} = bkClimb(el.closest(".bk-block"))
            {useSlot} = false
            @(sl)[
              @(bl)[ useSlot = bl.contains(sl) ][ useSlot = true ]
            ]
            @(useSlot)[
              t.kind = "slot"
              t.el = sl
              t.owner = sl.dataset.owner
              t.slot = number(sl.dataset.slot)
            ][
              @(bl)[
                {bshape} = bkShape(bl.dataset.type)
                {r} = bl.getBoundingClientRect()
                t.el = bl
                t.id = bl.dataset.id
                t.owner = bl.dataset.id
                t.kind = "after"
                @(or(eq(bshape, "wrap"), eq(bshape, "hat")))[
                  {hr} = bl.firstElementChild.getBoundingClientRect()
                  @(lt(y, add(hr.top, div(hr.height, 2))))[
                    @(neq(bshape, "hat"))[ t.kind = "before" ][ t.kind = "slotstart" ]
                  ][
                    @(lte(y, hr.bottom))[ t.kind = "slotstart" ]
                  ]
                ][
                  @(lt(y, add(r.top, div(r.height, 2))))[ t.kind = "before" ]
                ]
              ]
            ]
          ]
        ]
      ]
    ]
  ]
  t
]

#bkHideInd[ bkInd.style.display = "none" ]

#bkDragMove(x, y)[
  {g} = bkDrag
  g.ghost.style.left = join(sub(x, g.ox), "px")
  g.ghost.style.top = join(sub(y, g.oy), "px")
  {t} = bkTarget(x, y, get(g.moved, 0))
  @(g.hot)[ g.hot.classList.remove("bk-hot") ]
  g.hot = false
  @(eq(t.kind, "input"))[
    t.el.classList.add("bk-hot")
    g.hot = t.el
  ]
  bkPal.dataset.trash = "no"
  @(eq(t.kind, "trash"))[ bkPal.dataset.trash = "yes" ]
  bkShowInd(t)
  g.t = t
  // Near the canvas edge the view slides, so far-off places can be reached.
  {r} = bkCanvas.getBoundingClientRect()
  bkEdgeX = 0
  bkEdgeY = 0
  @(and(gte(y, r.top), lte(y, r.bottom), gte(x, r.left), lte(x, r.right)))[
    @(lt(x, add(r.left, 36)))[ bkEdgeX = 10 ]
    @(gt(x, sub(r.right, 36)))[ bkEdgeX = -10 ]
    @(lt(y, add(r.top, 36)))[ bkEdgeY = 10 ]
    @(gt(y, sub(r.bottom, 36)))[ bkEdgeY = -10 ]
  ]
  @(and(not(bkEdgeOn), or(neq(bkEdgeX, 0), neq(bkEdgeY, 0))))[
    bkEdgeOn = true
    bkEdgeStep()
  ]

// Picks the blocks up: they leave the canvas and ride along with the pointer.
#bkLift(x, y)[
  {c} = bkCand
  bkCand = false
  {liveEl} = c.el
  @(not(c.pal))[
    {fresh} = dom(join("#bkWorld [data-id=", c.id, "]"))
    @(fresh)[ liveEl = fresh ]
  ]
  {r} = liveEl.getBoundingClientRect()
  {moved} = list()
  @(c.pal)[
    {nb} = bkMake(bkWs, c.type)
    bkApplyFn(nb, c.fn)
    moved = list(nb)
  ][ moved = bkCut(c.id) ]
  @(size(moved))[
    {g} = map()
    g.moved = moved
    g.ox = sub(c.x, r.left)
    g.oy = sub(c.y, r.top)
    g.hot = false
    g.t = false
    g.ghost = bkMk("div", "bk-ghost")
    %(moved, b)[ g.ghost.appendChild(bkBlockEl(b, false)) ]
    g.ghost.style.transformOrigin = "0 0"
    @(not(c.pal))[ g.ghost.style.transform = join("scale(", bkWs.view.z, ")") ]
    doc().body.appendChild(g.ghost)
    bkDrag = g
    @(not(c.pal))[ bkRender() ]
    bkDragMove(x, y)
  ]
]

// Lets go: the blocks land where the pointer is.
#bkEnd(x, y)[
  {g} = bkDrag
  bkDrag = false
  bkEdgeX = 0
  bkEdgeY = 0
  g.ghost.remove()
  @(g.hot)[ g.hot.classList.remove("bk-hot") ]
  bkHideInd()
  bkPal.dataset.trash = "no"
  {first} = get(g.moved, 0)
  {t} = bkTarget(x, y, first)
  {k} = t.kind
  {done} = false
  @(eq(k, "trash"))[
    bkSel = ""
    done = true
  ]
  @(eq(k, "input"))[
    {owner} = bkFind(bkWs, t.owner)
    @(owner)[
      {old} = get(owner.args, t.arg)
      @(has(old, "b"))[ bkAddStack(list(old.b), add(bkWorldX(x), 24), add(bkWorldY(y), 24)) ]
      {na} = map()
      na.b = first
      put(owner.args, t.arg, na)
      done = true
    ]
  ]
  @(or(eq(k, "before"), eq(k, "after")))[
    {loc} = bkLocate(t.id)
    @(has(loc, "list"))[
      {at} = loc.idx
      @(eq(k, "after"))[ at = add(at, 1) ]
      bkSplice(loc.list, at, g.moved)
      done = true
    ]
  ]
  @(or(eq(k, "slot"), eq(k, "slotstart")))[
    {owner} = bkFind(bkWs, t.owner)
    @(owner)[
      {si} = 0
      @(eq(k, "slot"))[ si = t.slot ]
      {body} = get(owner.slots, si)
      {at} = size(body)
      @(eq(k, "slotstart"))[ at = 0 ]
      bkSplice(body, at, g.moved)
      done = true
    ]
  ]
  @(not(done))[
    // On the canvas (or let go somewhere else): a new stack where the blocks were carried to.
    {r} = bkCanvas.getBoundingClientRect()
    {px} = bkClamp(sub(x, g.ox), r.left, sub(r.right, 60))
    {py} = bkClamp(sub(y, g.oy), r.top, sub(r.bottom, 40))
    bkAddStack(g.moved, bkWorldX(px), bkWorldY(py))
  ]
  @(neq(k, "trash"))[ bkSel = first.id ]
  bkChanged()
]

    @(and(neq(tag, "INPUT"), neq(tag, "TEXTAREA")))[
      @(or(eq(e.key, "Delete"), eq(e.key, "Backspace")))[
        e.preventDefault()
        bkDelete()
      ]
    ]
  ]
]

// ---------- errors ----------

// Outlines the block that made the game's error, and brings it into view the first time.
#bkMarkErr(force)[
  {key} = join(errFile, ":", errTab, ":", errLine, ":", gameSel)
  @(or(force, neq(key, bkErrSeen)))[
    bkErrSeen = key
    %(domAll("#bkWorld .bk-err"), e)[ e.classList.remove("bk-err") ]
    @(and(bkWs, eq(errFile, "game"), eq(errTab, gameSel), gt(errLine, 0)))[
      {id} = bkBlockAtLine(bkWs, errLine)
      @(id)[
        {el} = dom(join("#bkWorld [data-id=", id, "]"))
        @(el)[
          el.classList.add("bk-err")
          @(neq(key, bkErrKey))[
            bkErrKey = key
            {r} = el.getBoundingClientRect()
            {cr} = bkCanvas.getBoundingClientRect()
            @(or(lt(r.top, cr.top), gt(r.bottom, cr.bottom), lt(r.left, cr.left), gt(r.right, cr.right)))[
              bkWs.view.x = add(bkWs.view.x, sub(add(cr.left, div(cr.width, 2)), add(r.left, div(r.width, 2))))
              bkWs.view.y = add(bkWs.view.y, sub(add(cr.top, div(cr.height, 2)), add(r.top, div(r.height, 2))))
              bkApplyView()
            ]
          ]
        ]
      ]
    ]
  ]
]

// ---------- keeping in step with Abyss Engine ----------

// Reads the blocks of the game tab that is open.
#bkLoad[
  {txt} = get(gameTexts, gameSel)
  bkWs = bkDecode(txt)
  bkWsIdx = gameSel
  bkWsText = txt
  bkSel = ""
  bkErrSeen = ""
  bkErrKey = ""
  bkRender()
]

#bkStep[
  {html} = doc().documentElement
  {on} = bkOn()
  // Packs that add blocks after this one has started are picked up here.
  @(neq(size(bkOrder), bkPalN))[
    bkPalN = size(bkOrder)
    bkRenderPal()
  ]
  html.classList.toggle("bk-on", on)
  html.classList.toggle("bk-game", and(on, eq(tab, "game")))
  @(on)[
    // The system tabs are hidden, so if one is showing, go back to the Game tab.
    @(has(list("engine", "compiler", "ui", "store", "packs"), tab))[ setTab("game") ]
    @(eq(tab, "game"))[
      {txt} = get(gameTexts, gameSel)
      @(or(not(bkWs), neq(bkWsIdx, gameSel), neq(txt, bkWsText)))[ bkLoad() ]
      bkMarkErr(false)
    ]
  ]
]

#bkTick[
  later(80, #[
    bkStep()
    bkTick()
  ])
]

// ---------- the File menu ----------

#bkShowMenu[
  {t} = "Off"
  @(bkSet.text)[ t = "On" ]
  bkMenuBtn.lastElementChild.textContent = t
]

#bkToggleText[
  bkSet.text = not(bkSet.text)
  bkSaveSet()
  bkShowMenu()
  @(eq(tab, "game"))[
    @(bkSet.text)[ showEditor() ][
      // Back to blocks. Anything typed in the editor that the blocks do not match is left behind.
      {cur} = get(gameTexts, gameSel)
      @(and(bkIsBlocksText(cur), neq(cur, bkEncode(bkDecode(cur)))))[ toast("Code typed in the editor was not turned into blocks.") ]
      bkWs = false
    ]
  ]
]
